NEW ARRAY claves[100];
NEW ARRAY precios[100];
NEW ARRAY stock[100];

SET cantidad TO 0;
SET opcion TO 0;

WHILE (opcion != 5) DO

    WRITE "===== INVENTARIO =====";
    WRITE "1. Agregar producto";
    WRITE "2. Mostrar productos";
    WRITE "3. Buscar producto";
    WRITE "4. Vender producto";
    WRITE "5. Salir";

    READ opcion;

    SWITCH opcion

    CASE 1

        WRITE "Clave del producto:";
        READ claves[cantidad];

        WRITE "Precio:";
        READ precios[cantidad];

        WRITE "Stock:";
        READ stock[cantidad];

        SET cantidad TO cantidad + 1;

        WRITE "Producto agregado";

    CASE 2

        IF cantidad == 0 THEN

            WRITE "Inventario vacio";

        ELSE

            FOR i = 0 TO cantidad - 1

                WRITE "Producto ", i;
                WRITE "Clave: ", claves[i];
                WRITE "Precio: ", precios[i];
                WRITE "Stock: ", stock[i];

            ENDFOR

        ENDIF

    CASE 3

        WRITE "Clave a buscar:";
        READ claveBuscar;

        SET encontrado TO 0;

        FOR i = 0 TO cantidad - 1

            IF claves[i] == claveBuscar THEN

                WRITE "Producto encontrado";
                WRITE "Precio: ", precios[i];
                WRITE "Stock: ", stock[i];

                SET encontrado TO 1;

            ENDIF

        ENDFOR

        IF encontrado == 0 THEN
            WRITE "Producto no encontrado";
        ENDIF

    CASE 4

        WRITE "Clave del producto:";
        READ claveVenta;

        WRITE "Cantidad a vender:";
        READ cantidadVenta;

        SET vendido TO 0;

        FOR i = 0 TO cantidad - 1

            IF claves[i] == claveVenta THEN

                IF stock[i] >= cantidadVenta THEN

                    SET stock[i] TO stock[i] - cantidadVenta;

                    WRITE "Venta realizada";
                    WRITE "Stock restante: ", stock[i];

                ELSE

                    WRITE "Stock insuficiente";

                ENDIF

                SET vendido TO 1;

            ENDIF

        ENDFOR

        IF vendido == 0 THEN
            WRITE "Producto inexistente";
        ENDIF

    CASE 5

        WRITE "Saliendo del sistema";

    DEFAULT_CASE

        WRITE "Opcion invalida";

    ENDSWITCH

ENDWHILE
