NEW ARRAY calificaciones[100];

SET suma TO 0.0;
SET mayor TO -1.0;
SET menor TO 101.0;

WRITE "Cantidad de alumnos:";
READ cantidad;

FOR i = 0 TO cantidad - 1

    WRITE "Calificacion del alumno ", i;
    READ calificaciones[i];

    SET suma TO suma + calificaciones[i];

    IF calificaciones[i] > mayor THEN
        SET mayor TO calificaciones[i];
    ENDIF

    IF calificaciones[i] < menor THEN
        SET menor TO calificaciones[i];
    ENDIF

ENDFOR

SET promedio TO suma / cantidad;

WRITE "Promedio general: ", promedio;
WRITE "Mayor calificacion: ", mayor;
WRITE "Menor calificacion: ", menor;

FOR j = 0 TO cantidad - 1

    IF calificaciones[j] >= 70.0 THEN
        WRITE "Alumno ", j, " APROBADO";
    ELSE
        WRITE "Alumno ", j, " REPROBADO";
    ENDIF

ENDFOR
